```c++
#include <iostream>
#include <string>
using namespace std;

int main() {
    string str;
    cin >> str;
    int N;
    cin >> N;
    
    for (int i = 0; i < N; i++) {
        int start, end;
        string front, back;
        cin >> start >> end >> front >> back;
        
        // 1. 剪切
        string clipboard = str.substr(start - 1, end - start + 1);
        str.erase(start - 1, end - start + 1);
        
        // 2. 粘贴
        string target = front + back;
        size_t pos = str.find(target);
        if (pos != string::npos) {
            // 找到，插在 front 后面
            pos += front.length();
        } else {
            // 没找到，插在最后
            pos = str.length();
        }
        str.insert(pos, clipboard);
        // 操作完成后剪贴板应被清空（题目要求），虽然下次循环会被覆盖，但逻辑上可体现。
    }
    
    cout << str << endl;
    return 0;
}
```
